 ), elected as (

 where e.rn=1
   and (case $1->>'date_basis' when 'report_date' then e.report_date else e.completed_date end)

   and ($1->>'agency_key' is null or e.agency_key=$1->>'agency_key')
   and ($1->>'manager_key' is null or e.manager_key=$1->>'manager_key')
